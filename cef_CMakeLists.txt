cmake_minimum_required(VERSION 3.10)

project(libHUI)

# Set C++ standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Set the CEF root directory
set(CEF_ROOT "_build_cache_cef_binary")

# Include the CEF macros and settings
find_package(CEF)

# Set output directories
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

# Add source files
#  we are at hui/build
set(SOURCE_FILES ../hui_webview__blink_cef.cc ../hui_windowcontrols__dummy.cc)

# Add executable target
#add_executable(PlibHUI ${SOURCE_FILES})
add_library(libHUI SHARED ${SOURCE_FILES})

# Link to the CEF lib wrapper
add_subdirectory(${CEF_ROOT} libcef_dll_wrapper)

# Link to CEF libraries
target_link_libraries(libHUI libcef_dll_wrapper)
#  we are at hui/build/build (cd must have happened somewhere before)

# linux: libcef.so looks up libc functions (close, localtime_r, getaddrinfo, ...) with dlsym(RTLD_NEXT), which fails when libcef is loaded after libc, so a tiny shim lib forwards those names to libc. It must be a DT_NEEDED of libHUI listed after libcef.so (link order), kept by --no-as-needed, and needs an explicit soname so libHUI records the plain name and finds it next to itself via $ORIGIN (not a build-machine path).
if(UNIX)
    add_library(cef_shim_linux SHARED ../cef_shim_linux.cc)
    set_target_properties(cef_shim_linux PROPERTIES PREFIX "" OUTPUT_NAME "cef_shim_linux" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    target_link_libraries(cef_shim_linux PRIVATE ${CMAKE_DL_LIBS})
    target_link_options(cef_shim_linux PRIVATE -Wl,-soname,cef_shim_linux.so)
    target_link_libraries(libHUI ${CMAKE_CURRENT_SOURCE_DIR}/_build_cache_cef_binary/Release/libcef.so)
    target_link_libraries(libHUI cef_shim_linux)
    target_link_options(libHUI PRIVATE -Wl,--no-as-needed)
endif()
if(WIN32)
	target_link_libraries(libHUI ../_build_cache_cef_binary/Release/libcef)
	#target_link_libraries(libHUI ../_build_cache_cef_binary/Release/libcef.lib)
endif()

# Use static runtime (MT) for all builds -- TODO: is this needed?
foreach(flag_var
    CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_DEBUG CMAKE_CXX_FLAGS_RELEASE
    CMAKE_C_FLAGS CMAKE_C_FLAGS_DEBUG CMAKE_C_FLAGS_RELEASE)
    if(${flag_var} MATCHES "/MD")
        string(REGEX REPLACE "/MD" "/MT" ${flag_var} "${${flag_var}}")
    endif()
    if(${flag_var} MATCHES "/MDd")
        string(REGEX REPLACE "/MDd" "/MTd" ${flag_var} "${${flag_var}}")
    endif()
endforeach()

# Additional include and link directories
include_directories(../c)
include_directories(../c++)
include_directories(${CEF_ROOT}/include)
include_directories(${CEF_ROOT})

# Platform-specific settings (windows)
if(WIN32)
    add_definitions(-D_WINDOWS)
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} /MANIFEST:NO")
endif()

# Platform-specific settings (linux)
if(UNIX)
    set_target_properties(libHUI PROPERTIES OUTPUT_NAME "HUI")
    set_target_properties(libHUI PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    set_target_properties(libHUI PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
endif()
